

CREATE TABLE sixty_min_mealbox_orders (
  id VARCHAR PRIMARY KEY DEFAULT gen_random_uuid(),
  
  -- Customer Information
  user_id VARCHAR,
  customer_name TEXT NOT NULL,
  customer_phone TEXT NOT NULL,
  customer_email TEXT,
  
  -- Delivery Address
  delivery_address TEXT NOT NULL,
  delivery_landmark TEXT,
  delivery_city TEXT DEFAULT 'Bangalore',
  delivery_pincode TEXT,
  delivery_latitude DECIMAL(10, 8),
  delivery_longitude DECIMAL(11, 8),
  
  -- Order Details
  order_number TEXT UNIQUE NOT NULL,
  box_template_id TEXT,
  box_template_name TEXT,
  portion_size INTEGER NOT NULL,
  box_quantity INTEGER NOT NULL DEFAULT 1,
  special_instructions TEXT,
  
  -- Dietary Configuration
  dietary_preference TEXT DEFAULT 'all',
  veg_count INTEGER DEFAULT 0,
  egg_count INTEGER DEFAULT 0,
  nonveg_count INTEGER DEFAULT 0,
  
  -- Timing
  requested_delivery_time TIMESTAMP,
  promised_delivery_by TIMESTAMP,
  order_placed_at TIMESTAMP DEFAULT NOW(),
  
  -- Financials
  per_box_price DECIMAL(10, 2) NOT NULL,
  subtotal DECIMAL(10, 2) NOT NULL,
  delivery_fee DECIMAL(10, 2) DEFAULT 0,
  tax_amount DECIMAL(10, 2) DEFAULT 0,
  discount_amount DECIMAL(10, 2) DEFAULT 0,
  discount_code TEXT,
  total_amount DECIMAL(10, 2) NOT NULL,
  
  -- Payment
  payment_method TEXT,
  payment_status TEXT DEFAULT 'pending',
  transaction_id TEXT,
  
  -- Order Status
  order_status TEXT DEFAULT 'pending',
  preparation_status TEXT DEFAULT 'pending',
  is_express BOOLEAN DEFAULT TRUE,
  
  -- Packaging
  include_utensils BOOLEAN DEFAULT TRUE,
  packaging_type TEXT DEFAULT 'mealbox',
  
  -- Audit
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);
-- Index for faster lookups
CREATE INDEX idx_sixty_min_mealbox_orders_user ON sixty_min_mealbox_orders(user_id);
CREATE INDEX idx_sixty_min_mealbox_orders_status ON sixty_min_mealbox_orders(order_status);
CREATE INDEX idx_sixty_min_mealbox_orders_date ON sixty_min_mealbox_orders(order_placed_at);

3. Shared Table for Order Items (Both Bulk & MealBox)
CREATE TABLE sixty_min_order_items (
  id VARCHAR PRIMARY KEY DEFAULT gen_random_uuid(),
  
  -- Link to parent order (one of these will be set)
  bulk_order_id VARCHAR REFERENCES sixty_min_bulk_orders(id) ON DELETE CASCADE,
  mealbox_order_id VARCHAR REFERENCES sixty_min_mealbox_orders(id) ON DELETE CASCADE,
  
  -- Item Details
  dish_id TEXT NOT NULL,
  dish_name TEXT NOT NULL,
  dish_category TEXT,
  dish_type TEXT,
  
  -- Quantity & Pricing
  quantity INTEGER NOT NULL,
  unit_price DECIMAL(10, 2) NOT NULL,
  total_price DECIMAL(10, 2) NOT NULL,
  
  -- Dietary Info
  dietary_type TEXT,
  is_spicy BOOLEAN DEFAULT FALSE,
  spice_level INTEGER DEFAULT 1,
  
  -- Customization
  special_instructions TEXT,
  
  -- Audit
  created_at TIMESTAMP DEFAULT NOW()
);
